cmake_minimum_required(VERSION 3.20)
project(openglaido_stt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_ENABLE_VALIDATION_HOOKS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRANSCRIBE_SOURCE}" transcribe)
# This pinned upstream revision assumes it is the top-level CMake project.
target_include_directories(transcribe PUBLIC "${TRANSCRIBE_SOURCE}/include")
add_executable(openglaido-stt main.cpp)
target_link_libraries(openglaido-stt PRIVATE transcribe)
target_compile_options(openglaido-stt PRIVATE -Wall -Wextra -Werror)
