cmake_minimum_required(VERSION 3.20)
project(openglaido_vibe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${VIBE_SOURCE}/src/asr_server.cpp")
    message(FATAL_ERROR "VIBE_SOURCE must point to the verified Microsoft VibeASR.cpp source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/patch-source.py"
    "${VIBE_SOURCE}/src/asr_server.cpp" "${CMAKE_CURRENT_BINARY_DIR}/vibe-server.cpp"
    COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${VIBE_SOURCE}" upstream EXCLUDE_FROM_ALL)
set_property(TARGET asr_stream_server PROPERTY SOURCES
    "${CMAKE_CURRENT_BINARY_DIR}/vibe-server.cpp" "${VIBE_SOURCE}/src/vae.cpp")
target_include_directories(asr_stream_server PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(asr_stream_server PROPERTIES OUTPUT_NAME openglaido-vibe)
